# SmartGate web app settings.
# Copy this file to ".env" in the same folder and fill in the values.
# Docker Compose reads ".env" automatically. Never commit ".env" (it holds secrets).

# "development" shows the DUMMY LOGIN button and reloads HTML on every request.
# Use "production" on the Jetson.
ENV=development

# Signs the login cookie. Use a long random value (at least 32 characters), e.g. from:
#   node -e "console.log(require('crypto').randomBytes(32).toString('hex'))"
# If empty, a random one is made on each start (everyone is logged out on restart).
SESSION_SECRET=

# GitHub login. Create an OAuth app at https://github.com/settings/developers
# with the callback URL http://<this-machine>:8000/auth
# If empty, the LOGIN button explains that GitHub login is not set up.
GITHUB_CLIENT_ID=
GITHUB_CLIENT_SECRET=

# Comma-separated GitHub usernames that get the admin role when they log in.
# Admins can open/close gates, change roles, remove users and remove alerts.
# Add "Dummy" to make the DUMMY LOGIN user an admin while testing.
ADMIN_USERS=Dummy

# Address of the Jetson's live monitoring service (src/main/live_api.py), used by the
# LIVE STREAM page. Its port is "server.live_port" in config/config.json (8081 by default).
# Example: JETSON_LIVE_URL=http://<jetson-ip-address>:8081
# If the web app runs in Docker on the same Jetson, use host.docker.internal:
#   JETSON_LIVE_URL=http://host.docker.internal:8081
# If empty, the LIVE STREAM page shows the Jetson as disconnected.
JETSON_LIVE_URL=http://137.92.214.122:8080

# Which Jetson service JETSON_LIVE_URL points to:
#   live_api    - the live monitoring service above (port 8081): camera, detections, system and
#                 sensor readings.
#   http_server - the Jetson's original web page (src/main/http_server.py, port 8080), for a Jetson
#                 without the live service. Shows the camera, CPU / memory / temperature and the gate
#                 position only; detections and PIR / Hall values show as "Unavailable". Example:
#                   JETSON_LIVE_URL=http://<jetson-ip-address>:8080
#                   JETSON_LIVE_SERVICE=http_server
#   smartgate_nano - the smartgate-nano project's dashboard server (port 8080). Shows everything:
#                 camera, gate state, Auto / Manual mode, state machine, detections, Jetson health
#                 (CPU / GPU / memory / disk), snapshots and the Jetson's own analytics. Example:
#                   JETSON_LIVE_URL=http://<jetson-ip-address>:8080
#                   JETSON_LIVE_SERVICE=smartgate_nano
#                 Check which one the Jetson runs: http://<jetson-ip>:8080/api/status answers with
#                 JSON on smartgate-nano only.
# If empty, live_api is used.
# With http_server or smartgate_nano, the LIVE STREAM page also has Open / Stop / Close gate buttons
# (admins only), and the Jetson's animal sightings and gate movements are saved for the dashboard
# and stats. On http_server, detections and stop need the Jetson update in
# setup/update_running_container.sh (not needed, and not for, smartgate-nano).
JETSON_LIVE_SERVICE=http_server

# smartgate_nano only: if the Jetson's config.json sets "server": { "token": ... }, put the same
# token here so the gate buttons work. Leave empty if it has no token.
JETSON_TOKEN=

# The gate number the Jetson's gate is recorded as on the dashboard and stats pages. If empty, 1 is used.
JETSON_GATE_NO=1
